// --- Entrance NPC: Central Prontera ---
prontera,107,190,4	script	Nightwing	10297,{
	mes "[Nightwing]";
	mes "The shadow of the Royal Nightmare looms over Prontera...";

	if (BaseLevel < 230) {
		mes "^FF0000You must be at least Base Level 230 to enter this realm.^000000";
		close;
	}

	// Re-enter (character bound)
	if ($@RN_Active && $@RN_PlayerID == getcharid(0) && gettimetick(2) < $@RN_TimeLimit) {
		.@rem = $@RN_TimeLimit - gettimetick(2);
		.@mins = .@rem / 60;
		.@secs = .@rem % 60;
		mes "^00FF00Your instance is still active! Time remaining: " + .@mins + "m " + .@secs + "s.^000000";
		next;
		if (select("Re-enter Instance:Cancel") == 1) {
			warp "prt_q", 154, 184;
			end;
		}
		close;
	}

	// 8 hour cooldown (character variable)
	if (RN_Cooldown > gettimetick(2)) {
		.@cd = RN_Cooldown - gettimetick(2);
		mes "^FF0000The Nightmare is still recovering from your last visit.^000000";
		mes "Come back in ^0000FF" + (.@cd / 3600) + "h " + ((.@cd % 3600) / 60) + "m^000000.";
		close;
	}

	next;
	mes "[Nightwing]";
	mes "On the night of the royal banquet, the crown of Rune-Midgarts fell without a single sword being drawn. The nobles raised their glasses... and by dawn, not one of them was left to lower it.";
	next;
	mes "[Nightwing]";
	mes "A sorceress called the Nightmare Queen seized the throne in the chaos. Her Death Witches wove a curse through the very wine of the feast, and the banquet never ended. It became a nightmare with no waking.";
	next;
	mes "[Nightwing]";
	mes "Prontera as you know it is gone. In its place stands a twisted mirror of the city, where every guest, guard and merchant has been turned into a monster.";
	next;
	mes "[Nightwing]";
	mes "I can open a door into that place for you. Few who have gone in have come back out as themselves.";
	next;
	mes "[Nightwing]";
	mes "Are you prepared to face the twisted horrors of the fallen court?";
	next;

	if (select("Enter Royal Nightmare:Cancel") == 2) close;

	if ($@RN_Active) {
		mes "[Nightwing]";
		mes "Another adventurer is currently inside the Nightmare. The door will not open for two souls at once. Please try again shortly.";
		close;
	}

	mes "[Nightwing]";
	mes "Then step through. Remember: inside, I serve her. Do not forget that I am only pretending.";
	close2;

	$@RN_Active = 1;
	$@RN_PlayerID = getcharid(0);	// character id (instance is character bound)
	$@RN_AID = getcharid(3);	// account id, only used to attach the HUD
	$@RN_Stage = 0;
	$@RN_Phase = 0;
	$@RN_Goal = 500;
	$@RN_BossKilled = 0;
	$@RN_BossNeed = 0;
	$@RN_TimeLimit = gettimetick(2) + 3600;

	// Remove every non-instance NPC from the map
	donpcevent "RN_NpcCleaner::OnClear";

	warp "prt_q", 154, 184;

	mapannounce "prt_q", "Instance: Royal Nightmare | Player: " + strcharinfo(0) + " | Time Limit: 60 Minutes", bc_map, "0x00FFCC";

	initnpctimer "RN_Core_Engine";
	end;
}

// --- Engine: waves, kill events, live HUD, time limit, cleanup ---
-	script	RN_Core_Engine	-1,{

// Starts a horde round: 700 of each monster at once
OnStartWave:
	callsub L_Cfg;
	if ($@RN_Phase == 1)
		$@RN_Goal = 500;
	else
		$@RN_Goal = 666;

	for (.@i = 0; .@i < 3; .@i++) {
		$@RN_K[.@i] = 0;
		areamonster "prt_q", 50, 50, 250, 250, .nm$[.@i], .id[.@i], .spawn, .ev$[.@i];
	}
	mapannounce "prt_q", "[Nightwing]: Horde " + $@RN_Phase + " - 2,100 Nightmarish Fiends have appeared! Slay " + $@RN_Goal + " of each kind!", bc_map, "0x00FF99";
	end;

OnGroteKill:
	callsub L_Kill, 0;
	end;

OnPierrotKill:
	callsub L_Kill, 1;
	end;

OnDisguiserKill:
	callsub L_Kill, 2;
	end;

L_Kill:
	.@t = getarg(0);
	if (!$@RN_Active || $@RN_Stage != 1) return;

	if ($@RN_K[.@t] < $@RN_Goal) $@RN_K[.@t]++;

	if (playerattached())
		dispbottom "[Royal Nightmare] Grote: " + $@RN_K[0] + "/" + $@RN_Goal + " | Pierrotzoist: " + $@RN_K[1] + "/" + $@RN_Goal + " | Disguiser: " + $@RN_K[2] + "/" + $@RN_Goal;

	if ($@RN_K[0] >= $@RN_Goal && $@RN_K[1] >= $@RN_Goal && $@RN_K[2] >= $@RN_Goal) {
		$@RN_Stage = 2;
		killmonster "prt_q", "All";	// clear the leftovers of the 700 spawn
		mapannounce "prt_q", "[Nightwing]: The horde has been purged! Speak to Nightwing at the center.", bc_map, "0xFFCC00";

		movenpc "Nightwing#RN", 158, 184;
		hideoffnpc "Nightwing#RN";
	}
	return;

// Boss kills (Death Witches / Demi Freyja)
OnQueenKill:
	if (!$@RN_Active || $@RN_Stage != 3) end;

	$@RN_BossKilled++;
	if (playerattached())
		dispbottom "[Royal Nightmare] Bosses Defeated: " + $@RN_BossKilled + "/" + $@RN_BossNeed;

	if ($@RN_BossKilled >= $@RN_BossNeed) {
		$@RN_Stage = 4;
		$@RN_FreyjaGID = 0;
		killmonster "prt_q", "All";	// clears any summoned adds
		mapannounce "prt_q", "[Royal Nightmare]: The Nightmare Queen's forces have been vanquished! Talk to Nightwing.", bc_map, "0x00FF00";

		movenpc "Nightwing#RN", 158, 184;
		hideoffnpc "Nightwing#RN";
	}
	end;

// 1-second loop: live HUD (time left + progress) and time limit check
OnTimer1000:
	if (!$@RN_Active) {
		stopnpctimer;
		end;
	}

	.@rem = $@RN_TimeLimit - gettimetick(2);

	if (.@rem <= 0) {
		stopnpctimer;
		mapannounce "prt_q", "[Royal Nightmare]: Time limit expired! You are being teleported out.", bc_map, "0xFF0000";
		sleep 2000;
		mapwarp "prt_q", "prontera", 150, 150;
		donpcevent "RN_Core_Engine::OnCleanup";
		end;
	}

	// Grand finale: every 15s, 10 of each horde monster appear around the final MVP
	if ($@RN_Stage == 3 && $@RN_Phase == 3 && gettimetick(2) >= $@RN_AddTime) {
		.@alive = 1;
		.@fx = 154;	// fallback position if the MVP's GID is unavailable
		.@fy = 185;
		if ($@RN_FreyjaGID) {
			if (getunitdata($@RN_FreyjaGID, .@fd) != -1) {
				.@fx = .@fd[UMOB_X];
				.@fy = .@fd[UMOB_Y];
			} else {
				.@alive = 0;	// final MVP is dead, stop summoning
				$@RN_FreyjaGID = 0;
				$@RN_AddTime = gettimetick(2) + 999999;
			}
		}
		if (.@alive) {
			callsub L_Cfg;
			for (.@j = 0; .@j < 3; .@j++)
				areamonster "prt_q", .@fx - 6, .@fy - 6, .@fx + 6, .@fy + 6, .nm$[.@j], .id[.@j], 10;
			mapannounce "prt_q", "[Royal Nightmare]: The Twisted God summons more Nightmarish Fiends!", bc_map, "0xFF6600";
			$@RN_AddTime = gettimetick(2) + 15;
		}
	}

	.@mins = .@rem / 60;
	.@secs = .@rem % 60;
	.@time$ = .@mins + ":" + (.@secs < 10 ? "0" : "") + .@secs;

	if ($@RN_Stage == 0)
		.@info$ = "Speak to Nightwing";
	else if ($@RN_Stage == 1)
		.@info$ = "Horde " + $@RN_Phase + " | G " + $@RN_K[0] + "/" + $@RN_Goal + " P " + $@RN_K[1] + "/" + $@RN_Goal + " D " + $@RN_K[2] + "/" + $@RN_Goal;
	else if ($@RN_Stage == 2 || $@RN_Stage == 4)
		.@info$ = "Speak to Nightwing";
	else if ($@RN_Stage == 3)
		.@info$ = "Bosses " + $@RN_BossKilled + "/" + $@RN_BossNeed;
	else
		.@info$ = "Claim your reward!";

	if (attachrid($@RN_AID)) {
		showscript "[Royal Nightmare] Time " + .@time$ + " | " + .@info$;
		if (.@rem == 600 || .@rem == 300 || .@rem == 60)
			dispbottom "[Royal Nightmare] " + (.@rem / 60) + " minute(s) remaining!";
		detachrid;
	}

	setnpctimer 0;
	end;

OnCleanup:
	stopnpctimer;
	killmonster "prt_q", "All";

	movenpc "Nightwing#RN", 158, 184;
	hideoffnpc "Nightwing#RN";
	hideonnpc "Nightmare Queen#RN";
	hideonnpc "Nightmare Chest#RN";

	$@RN_Active = 0;
	$@RN_PlayerID = 0;
	$@RN_AID = 0;
	$@RN_Stage = 0;
	$@RN_Phase = 0;
	$@RN_Goal = 500;
	$@RN_BossKilled = 0;
	$@RN_BossNeed = 0;
	$@RN_FreyjaGID = 0;
	$@RN_AddTime = 0;
	$@RN_TimeLimit = 0;
	deletearray $@RN_K[0], 3;
	end;

// Config is loaded on every use, so nothing depends on OnInit having run
L_Cfg:
	setarray .id[0], 20941, 20942, 20936;
	setarray .nm$[0], "Grote", "Pierrotzoist", "Disguiser";
	setarray .ev$[0], "RN_Core_Engine::OnGroteKill", "RN_Core_Engine::OnPierrotKill", "RN_Core_Engine::OnDisguiserKill";
	.spawn = 700;	// monsters spawned per type at the start of each horde
	return;
}

// --- Separate NPC so a problem here can never stop the engine from loading ---
-	script	RN_NpcCleaner	-1,{
OnClear:
	.@n = getunits(BL_NPC, .@gid, false, "prt_q");
	for (.@i = 0; .@i < .@n; .@i++) {
		.@nm$ = getunitname(.@gid[.@i]);
		if (.@nm$ == "Nightwing" || .@nm$ == "Nightmare Queen" || .@nm$ == "Nightmare Chest")
			continue;
		unitexit .@gid[.@i];
	}
	end;
}
