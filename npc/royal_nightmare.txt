// --- Instance BGM ---------------------------------------------------------
// Plays the Royal Nightmare theme for the instance player while they're on prt_q.
// The loadevent mapflag above fires this on every map load, including the warps
// between stages (a warp reloads the map, which resets the music to the map default,
// so the theme has to be started again each time). Leaving prt_q goes back to the
// destination map's normal BGM on its own.
// File: BGM/rn_royal_nightmare.mp3 inside your mod (name below, no folder, no .mp3).
-	script	RN_BGM	-1,{
OnPCLoadMapEvent:
	if (strcharinfo(3) != "prt_q") end;
	if (.debug) dispbottom "[RN BGM] map load on prt_q | active=" + $@RN_Active + " | you=" + getcharid(0) + " | owner=" + $@RN_PlayerID, 0xFFFF00;
	if (!$@RN_Active || !callfunc("RN_InRun")) end;
	playBGM "rn_royal_nightmare";
	// The bundled client drops playBGM's packet (0x0b8c), so client/index.js plays the
	// track when it sees this line. @@event lines are hidden from the chat box.
	dispbottom "@@event rn_bgm play";
	if (.debug) dispbottom "[RN BGM] playBGM + @@event sent: rn_royal_nightmare", 0x00FF00;
	end;

OnInit:
	.debug = 0;	// 1 = show [RN BGM] trace lines in chat
	end;
}

// --- Castle courtyard barricades (West / South / East gates) -------------
// The courtyard is sealed with a closed U-shaped wall, 2 cells thick, that blocks walking AND shooting:
//   West  : x 142..143, y 308..340  (from the south gate up to the river)
//   East  : x 167..168, y 308..340
//   South : x 142..168, y 308..309  (joins both columns, so the corners are closed)
// setcell is used instead of setwall: setwall stops at the first pillar/wall it meets,
// which is what left the gaps at the ends of each gate. setcell fills every cell.
// Failsafe: the engine timer also warps the instance player back out if they ever end up inside.
-	script	RN_Barricades	-1,{
OnInit:
	setarray .@x1[0], 142, 167, 142;	// West, East, South
	setarray .@y1[0], 308, 308, 308;
	setarray .@x2[0], 143, 168, 168;
	setarray .@y2[0], 340, 340, 309;
	for (.@i = 0; .@i < 3; .@i++) {
		setcell "prt_q", .@x1[.@i], .@y1[.@i], .@x2[.@i], .@y2[.@i], cell_walkable, false;
		setcell "prt_q", .@x1[.@i], .@y1[.@i], .@x2[.@i], .@y2[.@i], cell_shootable, false;
	}
	end;
}

// --- Party helpers ----------------------------------------------------------

// 1 when the attached character belongs to the open run: its owner, or a
// member of the owner's party who has joined it (talked to Nightwing)
function	script	RN_InRun	{
	if (getcharid(0) == $@RN_PlayerID) return 1;
	return ($@RN_PID && getcharid(1) == $@RN_PID && RN_Run == $@RN_RunID);
}

// The run's members who are on prt_q right now. Fills $@RN_MA[] (account
// ids) and returns how many.
function	script	RN_Members	{
	deletearray $@RN_MA[0], 128;
	.@n = 0;
	if ($@RN_PID) {
		getpartymember $@RN_PID, 1;
		copyarray .@cid[0], $@partymembercid[0], $@partymembercount;
		getpartymember $@RN_PID, 2;
		copyarray .@aid[0], $@partymemberaid[0], $@partymembercount;
		.@total = $@partymembercount;
		for (.@i = 0; .@i < .@total; .@i++) {
			if (!isloggedin(.@aid[.@i], .@cid[.@i])) continue;
			if (strcharinfo(3, .@cid[.@i]) != "prt_q") continue;
			$@RN_MA[.@n] = .@aid[.@i];
			.@n++;
		}
	} else if (isloggedin($@RN_AID, $@RN_PlayerID) && strcharinfo(3, $@RN_PlayerID) == "prt_q") {
		$@RN_MA[0] = $@RN_AID;
		.@n = 1;
	}
	return .@n;
}

// Only the run's owner (party leader / solo player) drives the story.
// Returns 1 for them; their party members are told so, anyone else ignored.
function	script	RN_IsOwner	{
	if (getcharid(0) == $@RN_PlayerID) return 1;
	if ($@RN_PID && getcharid(1) == $@RN_PID) {
		mes "[" + getarg(0) + "]";
		mes "Only your party leader, ^0000FF" + $@RN_Name$ + "^000000, can speak for the party in here.";
		close2;
	}
	return 0;
}

// The Queen's hoard for the attached character, and their 8 hour cooldown
function	script	RN_GiveRewards	{
	getitem 102071, 5;
	getitem 102074, 5;
	getitem 1000274, 30;
	getitem 102562, 3;	// Demi_Freyja_Antiquity
	Zeny += 5000000;
	getexp 100000000000, 60000000000;	// Base EXP, Job EXP
	RN_Cooldown = gettimetick(2) + 28800;	// 8 hour cooldown starts now (character bound)
	return;
}

// --- Portrait pop-up (cutin) helper ---------------------------------------
// callfunc "RN_Portrait", <id>;   1 = Nightwing (right), 2 = Nightmare Queen (left),
//                                  3 = Goddess of Nightmare (left), 0 = hide
// To use the OFFICIAL art instead of the bundled files, change the two names
// below to the official cutin filenames (see README, step 3).
// Files live in data/texture/ui/illust/ inside your mod.
function	script	RN_Portrait	{
	.@id = getarg(0);
	if (.@id == 1)      { .@f$ = "rn_nightwing.bmp"; .@pos = 2; }	// 4_M_SE_MOON (10297), bottom-right
	else if (.@id == 2) { .@f$ = "rn_leticia.bmp";   .@pos = 0; }	// 4_4JOB_LETICIA (10368), bottom-left
	else if (.@id == 3) { .@f$ = "rn_goddess.bmp";   .@pos = 0; }	// Goddess of Nightmare (Leticia's final form), bottom-left
	else { cutin "", 255; return; }
	cutin .@f$, .@pos;	// positions: 0 = bottom-left, 1 = bottom-middle, 2 = bottom-right
	return;
}

// --- Engine: waves, kill events, live HUD, time limit, cleanup ---
-	script	RN_Core_Engine	-1,{

// Starts a horde round: 700 of each monster at once
OnStartWave:
	callsub L_Cfg;
	if ($@RN_Phase == 1)
		$@RN_Goal = 500;
	else
		$@RN_Goal = 666;

	for (.@i = 0; .@i < 3; .@i++) {
		$@RN_K[.@i] = 0;
		areamonster "prt_q", 50, 50, 250, 250, .nm$[.@i], .id[.@i], .spawn, .ev$[.@i];
	}
	mapannounce "prt_q", "[Nightwing]: Horde " + $@RN_Phase + " - 2,100 Nightmarish Fiends have appeared! Slay " + $@RN_Goal + " each of " + .nm$[0] + ", " + .nm$[1] + " and " + .nm$[2] + "!", bc_map, "0x00FF99";
	end;

// Kill slot 0/1/2 of the current horde (see L_Cfg)
OnHordeKill0:
	callsub L_Kill, 0;
	end;

OnHordeKill1:
	callsub L_Kill, 1;
	end;

OnHordeKill2:
	callsub L_Kill, 2;
	end;

L_Kill:
	.@t = getarg(0);
	if (!$@RN_Active || $@RN_Stage != 1) return;

	if ($@RN_K[.@t] < $@RN_Goal) $@RN_K[.@t]++;

	callsub L_Cfg;
	if (playerattached())
		dispbottom "[Royal Nightmare] " + .nm$[0] + ": " + $@RN_K[0] + "/" + $@RN_Goal + " | " + .nm$[1] + ": " + $@RN_K[1] + "/" + $@RN_Goal + " | " + .nm$[2] + ": " + $@RN_K[2] + "/" + $@RN_Goal;

	if ($@RN_K[0] >= $@RN_Goal && $@RN_K[1] >= $@RN_Goal && $@RN_K[2] >= $@RN_Goal) {
		$@RN_Stage = 2;
		killmonster "prt_q", "All";	// clear the leftovers of the 700 spawn
		mapannounce "prt_q", "[Nightwing]: The horde has been purged! Speak to Nightwing at the center.", bc_map, "0xFFCC00";

		movenpc "Nightwing#RN", 158, 184;
		hideoffnpc "Nightwing#RN";
	}
	return;

// Boss kills (Death Witches / Demi Freyja)
OnQueenKill:
	if (!$@RN_Active || $@RN_Stage != 3) end;

	$@RN_BossKilled++;
	if (playerattached())
		dispbottom "[Royal Nightmare] Bosses Defeated: " + $@RN_BossKilled + "/" + $@RN_BossNeed;

	if ($@RN_BossKilled >= $@RN_BossNeed) {
		// 5 Death Witches down (after Horde 2): Horde 3, the Kades ambush
		if ($@RN_Phase == 2) {
			$@RN_Stage = 7;
			$@RN_KadesK = 0;
			$@RN_FlameK = 0;
			$@RN_FlameNeed = 15;

	areamonster "prt_q", 50, 50, 250, 250, "Kades", 2255, 5, "RN_Core_Engine::OnKadesKill";
	areamonster "prt_q", 50, 50, 250, 250, "--ja--", 22215, 8, "RN_Core_Engine::OnFlameKill";

// Horde 3, wave 2 (after the slap): two rows of 25 leading from 155,112 back to the
// central fountain, plus 20 spares scattered on the map (70 total). Horde 3 is over at 50 kills.
// Every beast spawns frozen (mode 0: it can neither move nor attack) and

	areamonster "prt_q", 50, 50, 250, 250, "--ja--", 20934, 10, "RN_Core_Engine::OnGauntletKill";

			setunitdata $@RN_GaunGID[.@i], UMOB_MODE, $@RN_GaunMode[.@i];
	mapannounce "prt_q", "[Royal Nightmare]: The Queen's beasts lunge!", bc_map, "0xFF0000";
	end;

// Freezes beast <gid> (its mode saved for OnGauntletWake) and turns it
// towards 155,112. Directions: 0 N, 1 NW, 2 W, 3 SW, 4 S, 5 SE, 6 E, 7 NE.
L_Freeze:
	.@g = getarg(0);
	// getunitdata is a command (it returns nothing when it works), so ask
	// unitexists whether the beast is there before reading its data
	if (!.@g || !unitexists(.@g)) return;
	getunitdata .@g, .@d;
	$@RN_GaunGID[$@RN_GaunN] = .@g;
	$@RN_GaunMode[$@RN_GaunN] = .@d[UMOB_MODE];
	$@RN_GaunN++;
	.@dx = 155 - .@d[UMOB_X];
	.@dy = 112 - .@d[UMOB_Y];
	.@ax = .@dx < 0 ? -.@dx : .@dx;
	.@ay = .@dy < 0 ? -.@dy : .@dy;
	if (.@ax * 2 < .@ay) .@dir = .@dy > 0 ? 0 : 4;
	else if (.@ay * 2 < .@ax) .@dir = .@dx > 0 ? 6 : 2;
	else if (.@dx > 0) .@dir = .@dy > 0 ? 7 : 5;
	else .@dir = .@dy > 0 ? 1 : 3;
	setunitdata .@g, UMOB_MODE, 0;
	setunitdata .@g, UMOB_DIR, .@dir;
	return;

OnGauntletKill:
	if (!$@RN_Active || $@RN_Stage != 9) end;
	$@RN_GauntK++;
	if (playerattached())
		dispbottom "[Royal Nightmare] Horde 3 | Queen's beasts: " + $@RN_GauntK + "/50";

		movenpc "Nightmare Queen#RN", 154, 184;
		hideoffnpc "Nightmare Queen#RN";
	}
	end;

	.@sy = getarg(1);
	if (checkcell("prt_q", .@sx, .@sy, cell_chkpass))
		monster "prt_q", .@sx, .@sy, getarg(2), getarg(3), 1, getarg(4);
	else
		areamonster "prt_q", .@sx - 2, .@sy - 2, .@sx + 2, .@sy + 2, getarg(2), getarg(3), 1, getarg(4);

// 1-second loop: live HUD (time left + progress) and time limit check
OnTimer1000:
	if (!$@RN_Active) {
		stopnpctimer;
		end;
	}

	.@rem = $@RN_TimeLimit - gettimetick(2);

	if (.@rem <= 0) {
		stopnpctimer;
		mapannounce "prt_q", "[Royal Nightmare]: Time limit expired! You are being teleported out.", bc_map, "0xFF0000";
		sleep 2000;
		mapwarp "prt_q", "prontera", 150, 150;
		donpcevent "RN_Core_Engine::OnCleanup";
		end;
	}

	// Grand finale: every 15s the Goddess calls a wave that SURROUNDS THE PLAYER:
	// 5 of each Horde 1 and Horde 2 monster (30), plus 1 Death Witch, 1 R001-Bestia,
	// 1 Bone Detardeurus, 1 Kades and 1 Mutated Tainted Flame Spirit (35 in all).
	if ($@RN_Stage == 3 && $@RN_Phase == 3 && gettimetick(2) >= $@RN_AddTime) {
		.@alive = 1;
		if ($@RN_FreyjaGID) {
			if (!unitexists($@RN_FreyjaGID)) {
				.@alive = 0;	// final MVP is dead, stop summoning
				$@RN_FreyjaGID = 0;
				$@RN_AddTime = gettimetick(2) + 999999;
			} else
				getunitdata $@RN_FreyjaGID, .@fd;
		}
		// She only calls them when she sees someone: a run member within 14
		// cells of her. Nobody near (e.g. out refilling after a death), or
		// 150+ monsters already out: no wave this time.
		.@near = 0;
		if (.@alive && $@RN_FreyjaGID && getmapunits(BL_MOB, "prt_q") < 150) {
			.@bx = .@fd[UMOB_X];
			.@by = .@fd[UMOB_Y];
			.@n = callfunc("RN_Members");
			copyarray .@aid[0], $@RN_MA[0], .@n;
			for (.@i = 0; .@i < .@n && !.@near; .@i++) {
				if (!attachrid(.@aid[.@i])) continue;
				getmapxy(.@pm$, .@px, .@py, BL_PC);
				if (.@pm$ == "prt_q" && distance(.@px, .@py, .@bx, .@by) <= 14) { .@near = 1; .@fx = .@px; .@fy = .@py; }
				detachrid;
			}
		}
		if (.@alive && !.@near) $@RN_AddTime = gettimetick(2) + 3;	// look again in a few seconds
		if (.@alive && .@near) {
			callsub L_Cfg;
			// No event labels on any of these: they must NOT count toward
			// OnQueenKill / RN_BossKilled, or the finale would end early.
			for (.@j = 0; .@j < 3; .@j++) {
				callsub L_SpawnRing, .@fx, .@fy, 5, .h1_nm$[.@j], .h1_id[.@j], 5, "";
				callsub L_SpawnRing, .@fx, .@fy, 5, .h2_nm$[.@j], .h2_id[.@j], 5, "";
			}
			callsub L_SpawnRing, .@fx, .@fy, 6, "Death Witch", 20943, 1, "";
			callsub L_SpawnRing, .@fx, .@fy, 6, "--ja--", 20934, 1, "";	// R001-Bestia
			callsub L_SpawnRing, .@fx, .@fy, 6, "--ja--", 20618, 1, "";	// Bone Detardeurus
			callsub L_SpawnRing, .@fx, .@fy, 6, "Kades", 2255, 1, "";
			callsub L_SpawnRing, .@fx, .@fy, 6, "--ja--", 22215, 1, "";	// Mutated Tainted Flame Spirit
			mapannounce "prt_q", "[Goddess of Nightmare]: Every soul you cut down in my court... rise, and surround the dreamer!", bc_map, "0xFF6600";
			$@RN_AddTime = gettimetick(2) + 15;
		}
	}

	// the beasts wake on their own if nobody closed the introduction in 30s
	if ($@RN_GaunWake && gettimetick(2) >= $@RN_GaunWake)
		donpcevent "RN_Core_Engine::OnGauntletWake";

	.@mins = .@rem / 60;
	.@secs = .@rem % 60;
	.@time$ = .@mins + ":" + (.@secs < 10 ? "0" : "") + .@secs;

	if ($@RN_Stage == 0)
		.@info$ = "Speak to Nightwing";
	else if ($@RN_Stage == 1) {
		callsub L_Cfg;
		.@info$ = "Horde " + $@RN_Phase + " | " + .ab$[0] + " " + $@RN_K[0] + "/" + $@RN_Goal + " " + .ab$[1] + " " + $@RN_K[1] + "/" + $@RN_Goal + " " + .ab$[2] + " " + $@RN_K[2] + "/" + $@RN_Goal;
	}
	else if ($@RN_Stage == 2 && $@RN_Phase == 3)
		.@info$ = "Face the Nightmare Queen";
	else if ($@RN_Stage == 2 || $@RN_Stage == 4 || $@RN_Stage == 8)
		.@info$ = "Speak to Nightwing";
	else if ($@RN_Stage == 7)
		.@info$ = "Horde 3 | Kades " + $@RN_KadesK + "/10 Flames " + $@RN_FlameK + "/" + $@RN_FlameNeed;
	else if ($@RN_Stage == 9)
		.@info$ = "Horde 3 | Beasts " + $@RN_GauntK + "/50";
	else if ($@RN_Stage == 3)
		.@info$ = "Bosses " + $@RN_BossKilled + "/" + $@RN_BossNeed;
	else
		.@info$ = "Claim your reward!";

	.@n = callfunc("RN_Members");
	copyarray .@aid[0], $@RN_MA[0], .@n;
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!attachrid(.@aid[.@i])) continue;
		// Failsafe: the courtyard behind the barricades is off-limits
		getmapxy(.@pm$, .@px, .@py, BL_PC);
		if (.@pm$ == "prt_q" && .@px >= 144 && .@px <= 166 && .@py >= 310) {
			warp "prt_q", 155, 307;
			dispbottom "[Royal Nightmare] The barricades force you back.";
		}
		showscript "[Royal Nightmare] Time " + .@time$ + " | " + .@info$;
		if (.@rem == 600 || .@rem == 300 || .@rem == 60)
			dispbottom "[Royal Nightmare] " + (.@rem / 60) + " minute(s) remaining!";
		detachrid;
	}

	setnpctimer 0;
	end;

// The rest of the party's share of the Queen's hoard (the owner got theirs

OnRewardParty:

	$@RN_FlameK = 0;

	$@RN_GauntK = 0;

		copyarray .nm$[0], .h2_nm$[0], 3;
		copyarray .ab$[0], .h2_ab$[0], 3;
	} else {
		copyarray .id[0], .h1_id[0], 3;
		copyarray .nm$[0], .h1_nm$[0], 3;
		copyarray .ab$[0], .h1_ab$[0], 3;
	}
	setarray .ev$[0], "RN_Core_Engine::OnHordeKill0", "RN_Core_Engine::OnHordeKill1", "RN_Core_Engine::OnHordeKill2";
	.spawn = 700;	// monsters spawned per type at the start of each horde
	return;

// Spawns <count> of one monster on a ring of <radius> cells around cx,cy.
// Consecutive calls keep walking around the ring, so mixed waves surround the target.
// callsub L_SpawnRing, <cx>, <cy>, <radius>, "<name>", <id>, <count>, "<event>";
L_SpawnRing:
	setarray .@cs[0], 1000,  924,  707,  383,    0, -383, -707, -924, -1000, -924, -707, -383,     0,  383,  707,  924;
	setarray .@sn[0],    0,  383,  707,  924, 1000,  924,  707,  383,     0, -383, -707, -924, -1000, -924, -707, -383;

		.@a = .ringpos % 16;
		.ringpos++;
		.@r = getarg(2) + rand(0, 2);
		callsub L_SpawnAt, getarg(0) + .@cs[.@a] * .@r / 1000, getarg(1) + .@sn[.@a] * .@r / 1000, getarg(3), getarg(4), getarg(6);
	}
	return;
}

// --- Separate NPC so a problem here can never stop the engine from loading ---
// Hides every other NPC on prt_q (the Sky Fortress entrance, the warp to
// prt_lib_q) while a run is on, and OnRestore (from OnCleanup) brings them back.
-	script	RN_NpcCleaner	-1,{
OnClear:
	callsub L_Restore;	// nothing left hidden from an earlier run
	.@n = getmapunits(BL_NPC, "prt_q", .@gid);
	for (.@i = 0; .@i < .@n; .@i++) {
		.@nm$ = rid2name(.@gid[.@i]);	// the unique name disablenpc needs
		if (compare(.@nm$, "#RN"))	// the instance's own NPCs
			continue;
		disablenpc .@nm$;
		.hidden$[getarraysize(.hidden$)] = .@nm$;
	}
	end;

OnRestore:
	callsub L_Restore;
	end;

L_Restore:
	for (.@i = 0; .@i < getarraysize(.hidden$); .@i++)
		enablenpc .hidden$[.@i];
	deletearray .hidden$;
	return;
}
